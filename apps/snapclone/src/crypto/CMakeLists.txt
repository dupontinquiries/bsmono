# ─── snapclone_crypto ─────────────────────────────────────────────────────────
# Crypto module — switches between the Qt placeholder and the Rust backend
# at compile time based on SNAPCLONE_RUST_CRYPTO.
#
# When SNAPCLONE_RUST_CRYPTO=OFF (default):
#   Compiles CryptoService.cpp which uses Qt HMAC-SHA256 + XOR stream.
#   This is intentionally a placeholder — see CryptoService.hpp for details.
#
# When SNAPCLONE_RUST_CRYPTO=ON:
#   Compiles RustCryptoService.cpp (the C++ bridge) and links the Rust
#   static library built by the snapclone_rust_crypto_build custom target.
#   The C header (include/crypto/snapclone_crypto_ffi.h) is generated by
#   cbindgen in the Rust build.rs.
# ─────────────────────────────────────────────────────────────────────────────

set(CRYPTO_SOURCES CryptoService.cpp)
set(CRYPTO_HEADERS
    ${PROJECT_SOURCE_DIR}/include/crypto/ICryptoService.hpp
    ${PROJECT_SOURCE_DIR}/include/crypto/CryptoService.hpp
)

if(SNAPCLONE_RUST_CRYPTO)
    list(APPEND CRYPTO_SOURCES RustCryptoService.cpp)
    list(APPEND CRYPTO_HEADERS
        ${PROJECT_SOURCE_DIR}/include/crypto/RustCryptoService.hpp)
endif()

add_library(snapclone_crypto STATIC
    ${CRYPTO_SOURCES}
    # Headers listed so AUTOMOC can detect Q_OBJECT.
    ${CRYPTO_HEADERS}
)

target_include_directories(snapclone_crypto
    PUBLIC
        ${PROJECT_SOURCE_DIR}/include
)

target_link_libraries(snapclone_crypto
    PUBLIC
        snapclone_warnings
        Qt6::Core
)

# Link the Rust static library when the backend is active.
# snapclone_rust_crypto_lib is the IMPORTED target defined in root CMakeLists.
if(SNAPCLONE_RUST_CRYPTO)
    target_link_libraries(snapclone_crypto PUBLIC snapclone_rust_crypto_lib)
    target_link_libraries(snapclone_crypto PUBLIC dl pthread)
endif()

# DHT discovery is now pure C++ (JamiAdapter_dht_impl.cpp + opendht C API).
# The Rust staticlib is only needed for the crypto backend.

set_target_properties(snapclone_crypto PROPERTIES AUTOMOC ON)
